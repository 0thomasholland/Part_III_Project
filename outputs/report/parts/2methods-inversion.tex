\subsection{Bayseian Inversion}


% Paragraph: Frame the inversion problem. We have a prior mu, a forward
% operator A, and observations y with noise covariance C_noise. The 
% posterior is Gaussian by conjugacy.

\begin{equation}
    p(x | y) \propto p(y | x) \, p(x)
\end{equation}

\begin{equation}
    \mu_{\text{post}} = \mu_{\text{prior}} + 
    \mathcal{C}_{\text{prior}} A^* 
    (A \mathcal{C}_{\text{prior}} A^* + \mathcal{C}_{\text{noise}})^{-1}
    (y - A\mu_{\text{prior}})
\end{equation}
